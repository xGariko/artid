\subsubsection{Oggetti Entità}

\arrayrulecolor{SlateGray} % Cambia il colore delle linee della tabella
\begin{tabularx}{\textwidth}{|c|X|}
\hline
\headerTabella{Oggetti Entità}{Descrizione}
\rigaTabella{Membro}{Rappresenta l'entit\`a Membro AFAM principale del sistema, contenente tutte le informazioni anagrafiche e credenziali.}
\rigaTabella{ArtID}{Rappresenta l'entit\`a ``ArtID'', contenitore dinamico di materiali su cui si fonda il sistema, contenente tutti i metadati dell'ArtID, come i suoi permessi di visualizzazione.}
\rigaTabella{Materiale}{Rappresenta il singolo materiale multimediale caricato dal Membro e contiene i suoi metadati.}
\rigaTabella{Certificato}{Rappresenta il documento formale caricato dal Membro AFAM volto a comprovare la partecipazione o il conseguimento di un titolo di studio o professionale e contiene i suoi metadati.}
\rigaTabella{Token}{Rappresenta la condivisione esterna e contiene le informazioni necessarie per tracciare le visualizzazioni, lo stato della condivisione, e il riferimento all'ArtID associato.}
\rigaTabella{Condivisione}{Rappresenta la condivisione interna e contiene i riferimenti al destinatario della condivisione e all'ArtID associato.}
\hline
\end{tabularx}
